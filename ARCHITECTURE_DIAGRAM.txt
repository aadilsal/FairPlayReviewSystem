# Main Function
detect_balls(
    frames_dir='...',
    model_path='...',
    enable_motion_prediction=True,  ◄─ Enable feature
    max_gap_frames=5,                ◄─ Max gap size
    prediction_confidence=0.3,       ◄─ Confidence for predictions
    use_kalman=False,                ◄─ Use Kalman filter
)

# Tracker Class
tracker = BallTracker(max_gap_frames=5, use_kalman=False)
tracker.update(frame_idx, detection)  ◄─ Feed detections
tracker.fill_gaps()                   ◄─ Fill missing frames
detections = tracker.get_all_detections()  ◄─ Get results

# Helper Functions
fill_detection_gaps(detections, max_gap_frames=5)  ◄─ Post-process
filter_ball_detections(detections, 'sports ball')  ◄─ Filter by class
group_detections_by_frame(detections)              ◄─ Group by frame

# Presets
config = get_config('balanced')  ◄─ Get preset configuration
detect_balls(..., **config)      ◄─ Apply preset
